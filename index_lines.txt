2640:       const isAllHiragana = /^[\u3040-\u309F]+$/.test(qNorm);
2641:       const isAllKatakana = /^[\u30A0-\u30FF]+$/.test(qNorm);

2647:         }

2651:       let maxScore = 0;
2652:       // If query is in Semantic Dict, we TRUST Semantic Search (Layer 2) to give the main score (up to 7.0).
2653:       // Layer 0/1 only gives a tiny boost (0.5) for exact matches to help sorting, but not enough to pass the 1.8 threshold alone.
2654:       // If query is NOT in Semantic Dict (e.g. "簡単", "レンジ", "フライパン"), Layer 0/1 gives a passing score (2.5).
2655:       let baseSubstringScore = isSemanticQuery ? 0.5 : 2.5;
2656: 
2657:       // --- Layer 0: Direct normalized substring match ---
2658:       if (tNorm.includes(qNorm)) {
2659:         maxScore = baseSubstringScore;
2660:       } else {
2661:         // --- Layer 1: Hiragana-unified substring match ---
2662:         // (Only safe for semantic queries OR queries > 2 chars, to avoid false positive particle matches)
2663:         const tHira = toHiragana(tNorm);
2664:         if (tHira.includes(qHira) && maxScore < baseSubstringScore - 0.1) {
2665:           maxScore = baseSubstringScore - 0.1;
2666:         }
2667:       }
2668: 
2669:       // --- Layer 2: Offline Category Semantic Search ---
2670:       if (window.FOOD_VECTORS) {
2671:         let maxSemanticSim = 0;
2672: 
2673:         // Strict matching for query vs dictionary word with Subsumption
2674:         const matchedWords = [];
2675:         for (const qWord in allQueryWords) {
2676:           const qWordHira = toHiragana(qWord);
2677: 
2678:           let idx = -1;
2679:           let matchLen = 0;
2680:           if (qNorm === qWord || qHira === qWordHira) {
2681:             idx = 0; matchLen = qNorm.length;
2682:           } else if (qWord.length >= 2 && qNorm.includes(qWord)) {
2683:             idx = qNorm.indexOf(qWord); matchLen = qWord.length;
2684:           } else if (qWord.length >= 2 && qHira.includes(qWordHira)) {
2685:             idx = qHira.indexOf(qWordHira); matchLen = qWordHira.length;
2686:           }
2687: 
2688:           if (idx !== -1) {
2689:             matchedWords.push({ word: qWord, start: idx, end: idx + matchLen });
2690:           }
2691:         }
2692: 
2693:         // Filter out matches that are strictly contained within another match
2694:         // (e.g. "にく" inside "にんにく")
2695:         const queryWordsToUse = [];
2696:         for (let i = 0; i < matchedWords.length; i++) {
2697:           let isSubsumed = false;
2698:           for (let j = 0; j < matchedWords.length; j++) {
2699:             if (i === j) continue;
2700:             if (matchedWords[i].start >= matchedWords[j].start &&
2701:               matchedWords[i].end <= matchedWords[j].end &&
2702:               (matchedWords[j].end - matchedWords[j].start) > (matchedWords[i].end - matchedWords[i].start)) {
2703:               isSubsumed = true;
2704:               break;
2705:             }
2706:           }
2707:           if (!isSubsumed) {
2708:             queryWordsToUse.push(matchedWords[i].word);
2709:           }
2710:         }
2711: 
2712:         // === Target Word Subsumption ===
2713:         // Pre-compute all valid dictionary words present in the target text
2714:         // to prevent generic substrings (e.g. "ねぎ") from being extracted out of
2715:         // specific compounds (e.g. "玉ねぎ") if they overlap in the exact same position.
2716:         const matchedTargetWords = [];
2717:         for (const tWord in window.FOOD_VECTORS) {
2718:           let searchIdx = 0;
2719:           while (true) {
2720:             searchIdx = tNorm.indexOf(tWord, searchIdx);
2721:             if (searchIdx === -1) break;
2722: 
2723:             let isValid = true;
2724:             const isKatakanaWord = /^[\u30A0-\u30FF]+$/.test(tWord);
2725:             if (isKatakanaWord) {
2726:               const prevChar = searchIdx > 0 ? tNorm[searchIdx - 1] : '';
2727:               const nextChar = searchIdx + tWord.length < tNorm.length ? tNorm[searchIdx + tWord.length] : '';
2728:               const isPrevKatakana = /[\u30A0-\u30FF]/.test(prevChar);
2729:               const isNextKatakana = /[\u30A0-\u30FF]/.test(nextChar);
2730: 
2731:               if (isPrevKatakana || isNextKatakana) {
2732:                 isValid = false;
2733:               }
2734:             }
2735: 
2736:             // Critical Heuristic: Prevent short Hiragana words (e.g. "うに", "いか", "す") 
2737:             // from being falsely extracted out of grammar or other words (e.g. "ように", "すいか", "します").
2738:             const isHiraganaWord = /^[\u3040-\u309F]+$/.test(tWord);
2739:             if (isHiraganaWord && tWord.length <= 2) {
2740:               const prevChar = searchIdx > 0 ? tNorm[searchIdx - 1] : '';
2741:               const isPrevHiragana = /[\u3040-\u309F]/.test(prevChar);
2742:               if (isPrevHiragana) {
2743:                 isValid = false;
2744:               }
2745:             }
2746: 
2747:             if (isValid) {
2748:               matchedTargetWords.push({ word: tWord, start: searchIdx, end: searchIdx + tWord.length });
2749:             }
2750:             searchIdx += 1;
2751:           }
2752:         }
2753: 
2754:         const validTargetWords = new Set();
2755:         for (let i = 0; i < matchedTargetWords.length; i++) {
2756:           let isSubsumed = false;
2757:           for (let j = 0; j < matchedTargetWords.length; j++) {
2758:             if (i === j) continue;
2759:             const wi = matchedTargetWords[i];
2760:             const wj = matchedTargetWords[j];
2761:             if (wi.start >= wj.start && wi.end <= wj.end && (wj.end - wj.start) > (wi.end - wi.start)) {
2762:               isSubsumed = true;
2763:               break;
2764:             }
2765:           }
2766:           if (!isSubsumed) {
2767:             validTargetWords.add(matchedTargetWords[i].word);
2768:           }
2769:         }
2770: 
2771:         // Compute max semantic similarity between any qWord and valid tWord
2772:         for (const qWord of queryWordsToUse) {
2773:           const qVec = allQueryWords[qWord];
2774:           for (const tWord of validTargetWords) {
2775:             const tVec = window.FOOD_VECTORS[tWord];
2776:             let dot = 0;
2777:             for (let i = 0; i < qVec.length; i++) {
2778:               dot += qVec[i] * tVec[i];
2779:             }
2780:             // Prevent Dimensional Explosion by capping the dot product at 1.0
2781:             if (dot > 1.0) dot = 1.0;
2782: 
2783:             if (dot > maxSemanticSim) maxSemanticSim = dot;
2784:           }
2785:         }
2786: 
2787:         // Exact category match (dot == 1.0) gives semanticScore = 7.0
2788:         if (maxSemanticSim >= 0.55) {
2789:           const semanticScore = (maxSemanticSim - 0.5) * 14.0;
2790:           maxScore = Math.max(maxScore, semanticScore);
2791:         }
2792:       }
2793: 
2794:       return maxScore >= 1.8 ? maxScore : 0;
2795:     }

2797:     function computeRecipeSemanticScore(recipe, queryStr) {
2798:       // 1. Convert common punctuation separators to spaces to prevent words from merging (e.g. "豚肉、玉ねぎ" -> "豚肉 玉ねぎ")
2799:       let normalizedQuery = queryStr.replace(/[\u3000・,\/、。｜\|]/g, ' ');
2800:       const rawKeywords = normalizedQuery.trim().split(/\s+/).filter(k => k.length > 0);
2801:       if (rawKeywords.length === 0) return 1.0;
2802: 
2803:       // 2. Filter out short, non-semantic hiragana noise particles (e.g. "と", "で", "の")
2804:       // because they would return score 0 and fail the entire AND-search.
2805:       const keywords = [];
2806:       const allQueryWords = { ...(window.FOOD_VECTORS || {}), ...(window.QUERY_VECTORS || {}) };
2807:       for (const kw of rawKeywords) {
2808:         const qNorm = normalizeJapaneseText(kw);
2809:         const qHira = toHiragana(qNorm);
2810:         const isSemantic = (qNorm in allQueryWords) || (qHira in allQueryWords);
2811:         const isAllHira = /^[\u3040-\u309F]+$/.test(qNorm);
2812: 
2813:         if (!isSemantic && isAllHira && qNorm.length <= 2) continue;
2814:         keywords.push(kw);
2815:       }
2816: 
2817:       // If all keywords were noise particles, fallback to the raw keywords to prevent empty array
2818:       const finalKeywords = keywords.length > 0 ? keywords : rawKeywords;
2819: 
2820:       let totalScore = 0;
2821: 
2822:       for (const kw of finalKeywords) {
2823:         let kwMaxScore = computeSearchMatchScore(kw, recipe.title) * 2.0;
2824: 
2825:         if (Array.isArray(recipe.ingredients)) {
2826:           for (const ing of recipe.ingredients) {
2827:             kwMaxScore = Math.max(kwMaxScore, computeSearchMatchScore(kw, ing) * 1.5);
2828:           }
2829:         }
2830: 
2831:         if (Array.isArray(recipe.steps)) {
2832:           for (const st of recipe.steps) {
2833:             const stText = typeof st === 'object' ? `${st.title || ''} ${st.text || st.desc || ''}` : String(st);
2834:             kwMaxScore = Math.max(kwMaxScore, computeSearchMatchScore(kw, stText) * 0.6);
2835:           }
2836:         }
2837: 
2838:         if (recipe.memo) {
2839:           kwMaxScore = Math.max(kwMaxScore, computeSearchMatchScore(kw, recipe.memo) * 0.8);
2840:         }
2841: 
2842:         if (kwMaxScore <= 0) return 0;
2843:         totalScore += kwMaxScore;
2844:       }
2845:       return totalScore;
2846:     }
